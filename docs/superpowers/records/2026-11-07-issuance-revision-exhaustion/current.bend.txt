import Base
import ./Issuance.bend as I
import ./IssuanceTests.bend as IT

def revised(state: I.State, rev: Nat) -> I.State:
  match state:
    case I.State{_, id, live, directory, placements, approvals, grants, seen}:
      I.State{rev, id, live, directory, placements, approvals, grants, seen}

def label(change: I.Change) -> String:
  match change:
    case I.Rejected{_, _}: "REJECTED"
    case I.Changed{_, _}: "CHANGED"

def run(+value: Nat) -> String:
  state = revised(IT.registered(), value)
  label(I.revoke(state, value, IT.write_authority(), 2n))

def parsed(value: Maybe<&2, Nat>) -> String:
  match value:
    case None{}: "BAD NUMBER"
    case Some{x}: run(x)

def main() -> IO(Unit):
  do IO<Unit>:
    value : String <- IO.try(String, IO.get_env("WRIT_ISSUANCE_TEST"))
    IO.print(parsed(Nat.read(value)))
