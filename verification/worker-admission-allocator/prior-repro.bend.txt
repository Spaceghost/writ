# Synthetic fixtures only: no native identity evidence.
def set_counter(state: P.State, next: Nat) -> P.State:
  match state:
    case P.State{revision, identity, profiles, bindings, workers, _, seals, publications}:
      P.State{revision, identity, profiles, bindings, workers, next, seals, publications}

def invalid_admitted(step: P.Step, expected: W.Ref) -> Bool:
  match step:
    case P.Admitted{+after, P.DeliverInputs{worker, ref} <> []}:
      Nat.is_eq(worker, 0n) && W.Ref.equal(ref, expected) && Bool.not(P.valid(after))
        && Nat.is_eq(P.next_worker(after), 1n)
    case _: False{}

def direct() -> Bool:
  +state = set_counter(PT.base(), 0n)
  P.valid(state) && invalid_admitted(PT.admit_on(state, PT.full_report(), PT.mediated_inventory()), PT.input_ref())

def set_joint_counter(state: LH.State, next: Nat) -> LH.State:
  match state:
    case LH.State{p, records, clock, owner}: LH.State{set_counter(p, next), records, clock, owner}

def issued(+w: RT.Sc) -> LH.State:
  +dispatched = {LH.JointAction.state(LH.joint_dispatch(JT.launched(w), 1n, 110n)) : LH.State}
  LH.JointAction.state(LH.joint_issue(dispatched, 1n, 4, 120n, 110n))

def joint_step(outcome: LH.Outcome) -> Bool:
  match outcome:
    case LH.Stepped{_, _, _, +step}:
      invalid_admitted(step, W.Ref{60, 1, 1, 1})
        && PT.effects_equal(LH.Outcome.effects(outcome), [P.DeliverInputs{0n, W.Ref{60, 1, 1, 1}}])
    case _: False{}

def joint(+w: RT.Sc) -> Bool:
  +state = set_joint_counter(issued(w), 0n)
  P.valid(LH.portable(state)) && LH.joint_commit_room(state)
    && joint_step(LH.dispatch(state, LH.FromBroker{LH.Barrier{1n, JT.fact(), W.encode_bytes(JT.guard_use())}},
      LH.Sample{110n}, V.fixture_authority(110n)))

def main() -> IO(Unit):
  do IO<Unit>:
    Test.check(direct(), "existing direct admission emits worker zero and invalid state")
    Test.check(joint(RT.sc_world()), "existing joint barrier emits worker zero and invalid state")
