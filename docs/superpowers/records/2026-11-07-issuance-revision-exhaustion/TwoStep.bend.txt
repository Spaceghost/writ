import Base
import ./Issuance.bend as I
import ./IssuanceTests.bend as IT

def revised(state: I.State, rev: Nat) -> I.State:
  match state:
    case I.State{_, id, live, directory, placements, approvals, grants, seen}:
      I.State{rev, id, live, directory, placements, approvals, grants, seen}

def label(change: I.Change) -> String:
  match change:
    case I.Rejected{_, _}: "REJECTED"
    case I.Changed{_, _}: "CHANGED"

def run(+max: Nat) -> IO(Unit):
  +near = Nat.sub(max, 1n)
  +state = revised(IT.registered(), near)
  +first = I.revoke(state, near, IT.write_authority(), 2n)
  +after = I.Change.state(first)
  do IO<Unit>:
    IO.print(label(first))
    IO.print(label(I.revoke(after, max, IT.source_authority(), 1n)))

def parsed(value: Maybe<&2, Nat>) -> IO(Unit):
  match value:
    case None{}: IO.die(Unit, 1, "BAD NUMBER")
    case Some{x}: run(x)

def main() -> IO(Unit):
  do IO<Unit>:
    value : String <- IO.try(String, IO.get_env("WRIT_ISSUANCE_TEST"))
    parsed(Nat.read(value))
