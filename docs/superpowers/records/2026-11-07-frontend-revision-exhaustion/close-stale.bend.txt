import Base
import ./Frontend.bend as F
import ./FrontendTests.bend as FT
import ./TransferTests.bend as TF

def label(change: F.Change) -> String:
  match change:
    case F.Rejected{_}: "REJECTED"
    case F.Changed{_, _}: "CHANGED"

def run(+value: Nat) -> String:
  state = {F.State{value, 2n, F.Handles{FT.raw_handle(1n, TF.plan()), F.NoHandles{}}} : F.State}
  label(F.close(state, 0n, FT.ref(1n)))

def parsed(value: Maybe<&2, Nat>) -> String:
  match value:
    case None{}: "BAD NUMBER"
    case Some{x}: run(x)

def main() -> IO(Unit):
  do IO<Unit>:
    value : String <- IO.try(String, IO.get_env("WRIT_FRONTEND_TEST"))
    IO.print(parsed(Nat.read(value)))
