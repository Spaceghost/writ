#!/usr/bin/env python3
"""Exercise observe.sh's input guard with a tripwire before any host probe."""
import os
from pathlib import Path
import subprocess
import tempfile

observe = Path('scripts/gpu-validation/observe.sh').resolve()
with tempfile.TemporaryDirectory(prefix='writ-gpu-prelaunch-') as temp:
    root = Path(temp)
    pathbin = root / 'bin'; pathbin.mkdir()
    marker = root / 'dirname-reached'
    stub = pathbin / 'dirname'
    stub.write_text('#!/bin/sh\nprintf reached > "' + str(marker) + '"\nexit 99\n')
    stub.chmod(0o700)
    env = dict(os.environ, PATH=str(pathbin), GV_RUN_DIR=str(root / 'run'))
    cases = ['+RTS', '-RTS', '--RTS', 'text+RTSsuffix', 'line\nbreak',
             'line\rbreak', 'text\n--- record v1 ---\n']
    for argument in cases:
        r = subprocess.run(['/bin/sh', str(observe), '/bin/true', 'on', argument],
                           env=env, text=True, capture_output=True, timeout=5)
        if r.returncode != 64 or 'argument contains a line break or RTS delimiter' not in r.stderr:
            raise SystemExit(f'guard failed for {argument!r}: {r.returncode} {r.stderr!r}')
        if marker.exists() or (root / 'run').exists():
            raise SystemExit(f'guard reached later setup for {argument!r}')
        print(f'{argument!r}: refused before setup')
    # A benign argument reaches the injected dirname tripwire; PATH contains no
    # GPU tool and the stub exits before a workload can be launched.
    subprocess.run(['/bin/sh', str(observe), '/bin/true', 'on', '123'],
                   env=env, text=True, capture_output=True, timeout=5)
    if not marker.exists() or (root / 'run').exists():
        raise SystemExit('benign-argument tripwire did not establish reachability')
    print('benign argument: reached setup tripwire; no workload launched')
