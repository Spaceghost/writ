This first pass reads the file's own description of itself, its compile-time bounds, and the component entry. Some Genode vocabulary first. A session is a connection one component opens to a service another component (or the kernel-side core) provides; the parent (here, init) routes each session request according to its configuration. A capability is an unforgeable reference to an object in another component: holding it lets you call that object, nothing more. A quota is the RAM and capability count a component was given by its parent; opening a session donates part of it. The Pd_session (protection domain) is the component's own session to core for its address space, RAM, and caps. An Entrypoint is a thread that receives RPCs and signals for objects it manages. The shim is linked into each Bend program, replaces libc's mmap/munmap/mprotect, and supplies the C functions behind the Genode.log/now/sleep/take/put effects. void Libc::Component::construct(Libc::Env &env)
{
    _heap.construct(env.ram(), env.rm());
    _pd.construct(env.pd_session_cap());

    _env = &env;   /* lookup helpers need it; entry points stay ENOSYS until resolved */
    if (!_resolve_libc()) {
        _libc_mmap = nullptr; _libc_munmap = nullptr; _libc_mprotect = nullptr;
        env.parent().exit(1);
        return;
    }

#if defined(WRIT_SHIM_FAULT_REGION_MAP)
    /* the "writ_fault" entrypoint (header, "Fault delivery", case 2) */
    _install_fault_handler();
#endif

    /* Writ channel service, only with a <writ_channel> config node */
    _channel_init(env);

    /* as repos/libports/src/lib/posix/construct.cc */
    Libc::with_libc([&] () {
        int argc = 0; char **argv = nullptr; char **envp = nullptr;
        populate_args_and_env(env, argc, argv, envp);
        environ = envp;
        if (char const *v = getenv("WRIT_MEM_REPORT"))
            if (v[0] == '1' && v[1] == 0) atexit(_report_at_exit);
        exit(main(argc, argv, envp));
    });
}
