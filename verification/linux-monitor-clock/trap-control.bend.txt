def run_decoded(decoded: W.DecodeResult, +state: LH.State) -> IO(Unit):

    case W.Decoded{message}:
      +result = LH.barrier_with(None{}, message, LT.linux_report(), LH.inventory_of(LT.full_fact()), state,

      IO.print(Nat.show(List.length(&2, P.Effect, LH.Outcome.effects(result))))
