cmake_minimum_required(VERSION 3.24)
project(writ_windows_native C)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This native binding requires Windows x64")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "The qualified build uses the Windows SDK and MSVC")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WINFSP_ROOT "C:/Program Files (x86)/WinFsp" CACHE PATH "Pinned WinFsp SDK root")
find_path(WINFSP_INCLUDE_DIR winfsp/winfsp.h PATHS "${WINFSP_ROOT}/inc" NO_DEFAULT_PATH REQUIRED)
find_library(WINFSP_LIBRARY NAMES winfsp-x64 PATHS "${WINFSP_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)

add_executable(writfs-windows writfs.c worker.c policy_pipe.c isolation.c)
target_include_directories(writfs-windows PRIVATE "${WINFSP_INCLUDE_DIR}")
target_compile_definitions(writfs-windows PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(writfs-windows PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-windows PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(writfs-windows PRIVATE "${WINFSP_LIBRARY}" advapi32 bcrypt fwpuclnt rpcrt4 userenv ole32 ws2_32 OneCoreUAP)

add_executable(writfs-probe probe.c isolation.c channels.c)
target_compile_definitions(writfs-probe PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(writfs-probe PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-probe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(writfs-probe PRIVATE advapi32 bcrypt ws2_32)

add_executable(writfs-transport-probe transport_probe.c policy_pipe.c isolation.c)
target_compile_definitions(writfs-transport-probe PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(writfs-transport-probe PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-transport-probe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(writfs-transport-probe PRIVATE advapi32)

# Standalone initialization observations: no mount, policy, data input or actor.
add_executable(writfs-init-diagnostic init_diagnostic.c init_worker.c isolation.c)
target_compile_definitions(writfs-init-diagnostic PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
target_compile_options(writfs-init-diagnostic PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-init-diagnostic PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(writfs-init-diagnostic PRIVATE advapi32 bcrypt fwpuclnt rpcrt4 userenv ole32 ws2_32 OneCoreUAP)

add_executable(writfs-init-report-test init_report_test.c)
target_compile_options(writfs-init-report-test PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-init-report-test PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
enable_testing()
add_test(NAME init-report-framing COMMAND writfs-init-report-test)

# Minimal child deliberately has no static USER32 or OLE32 imports.
add_executable(writfs-init-child init_child.c)

add_executable(writfs-pss-observer pss_observer.c pss_table.c pss_progress.c)

target_compile_options(writfs-pss-child PRIVATE /W4 /sdl /guard:cf)
target_link_options(writfs-pss-child PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)

target_compile_definitions(writfs-pss-init-control PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
