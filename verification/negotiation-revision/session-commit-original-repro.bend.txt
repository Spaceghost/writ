import Base
import ./Ledger.bend as L
import ./Negotiation.bend as N
import ./NegotiationTests.bend as NF
import ./Backplane.bend as B
import ./BackplaneTests.bend as BF
import ./TransferTests.bend as TF

def revised(old: N.Session, rev: Nat) -> N.Session:
  match old:
    case N.Session{_, request, stage, pinned, challenge, receipt}: N.Session{rev, request, stage, pinned, challenge, receipt}

def label(value: B.CommitResult) -> String:
  match value:
    case B.Committed{_, _, _}: "APPLIED"
    case B.Rejected{_, _}: "REJECTED"

def evaluate(max: Nat) -> String:
  label(B.commit(revised(NF.running(), max), 6n, 1n, NF.held_ledger(), BF.candidate(), BF.context()))

def parsed(value: Maybe<&2, Nat>) -> String:
  match value:
    case None{}: "BAD NUMBER"
    case Some{max}: evaluate(max)

def main() -> IO(Unit):
  do IO<Unit>:
    value : String <- IO.try(String, IO.get_env("WRIT_LEDGER_TEST"))
    IO.print(parsed(Nat.read(value)))
