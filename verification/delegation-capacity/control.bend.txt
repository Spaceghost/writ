# Synthetic model fixtures; no native identity or platform is observed.
def identity_counters(state: I.State, revision: Nat, slot: Nat) -> I.State:
  match state:
    case I.State{_, host, incarnation, _, sessions, entries}: I.State{revision, host, incarnation, slot, sessions, entries}

def host_revision(state: P.State, revision: Nat) -> P.State:
  match state:
    case P.State{_, identity, profiles, bindings, workers, next, seals, publications}:
      P.State{revision, identity, profiles, bindings, workers, next, seals, publications}

def delegate(+state: I.State, ref: W.Ref, target: Nat, rights: U32, lifetime: Nat) -> I.Change:
  I.delegate(state, I.revision(state), IT.session_a(), ref, target, rights, lifetime, IT.now())

def identity_fixture(+state: I.State, ref: W.Ref, target: Nat, rights: U32, lifetime: Nat) -> IO(Unit):

    Test.check(I.valid(state), "identity fixture valid")
    Test.check(Bool.not(I.Change.accepted(delegate(state, ref, target, rights, lifetime))), "delegation must deny")

    case Some{max}: mode(name, max)
    case None{}: IO.die(Unit, 2, "bad max")

def run(args: List<String>) -> IO(Unit):

    case [_, name]: parsed(name, Nat.read("281474976710655"))
    case _: IO.die(Unit, 2, "expected mode")
