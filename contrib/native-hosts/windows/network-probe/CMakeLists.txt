cmake_minimum_required(VERSION 3.24)
project(writ_windows_network_evidence C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
  message(FATAL_ERROR "Network evidence tests require Windows x64, MSVC and the Windows SDK")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(writfs-network-test ../network_probe_test.c)
add_executable(writfs-network-loopback ../network_probe_loopback.c ../network_probe.c)
foreach(target writfs-network-test writfs-network-loopback)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /sdl /guard:cf)
  target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
  target_link_libraries(${target} PRIVATE ws2_32)
endforeach()
enable_testing()
add_test(NAME network-api-failure-evidence COMMAND writfs-network-test)
add_test(NAME network-loopback-positive-controls COMMAND writfs-network-loopback)
set_tests_properties(network-api-failure-evidence network-loopback-positive-controls PROPERTIES TIMEOUT 30)
